Supabase Migration Setup

Copy the SQL schema and run it in your Supabase SQL Editor

SQL Schema

-- Create Roles and Users Tables
create table if not exists roles (
  role_id serial primary key,
  role_name text unique not null
);

create table if not exists users (
  user_id serial primary key,
  role_id int not null references roles(role_id),
  full_name text not null,
  email text unique not null,
  password_hash text not null,
  google_oauth_id text,
  class_number int,
  board text,
  streak_days int default 0,
  xp_points int default 0,
  is_approved int default 0,
  created_at timestamptz default now()
);

-- Create Subjects and Content Formats
create table if not exists subjects (
  subject_id serial primary key,
  subject_name text not null,
  class_number int not null
);

create table if not exists content_formats (
  format_id serial primary key,
  format_name text unique not null
);

-- Create Learning Materials and Progress
create table if not exists learning_materials (
  material_id serial primary key,
  subject_id int not null references subjects(subject_id) on delete cascade,
  format_id int not null references content_formats(format_id),
  title text not null,
  description text,
  file_url text not null,
  duration_lessons text,
  instructor_name text,
  created_at timestamptz default now()
);

create table if not exists student_progress (
  progress_id serial primary key,
  student_id int not null references users(user_id) on delete cascade,
  material_id int not null references learning_materials(material_id) on delete cascade,
  completion_percentage int default 0,
  is_completed boolean default false,
  updated_at timestamptz default now()
);

-- Seed Initial Data (Roles and Formats)
insert into roles (role_name) select * from (values ('Admin'),('Teacher'),('Student')) as v(name)
  on conflict (role_name) do nothing;

insert into content_formats (format_name) select * from (values ('E-Book'),('Audio Book'),('Video Content')) as v(name)
  on conflict (format_name) do nothing;
📋 Steps to Complete:
  1. Go to Supabase → SQL Editor → New query
  2. Click "📋 Copy SQL" button above
  3. Paste into the SQL Editor and click Execute
  4. Create a storage bucket named learning_materials (Supabase → Storage → New bucket)
  5. Set bucket visibility to Public (or setup signed URLs for private)
  6. Get your Supabase URL and Anon Key from Settings → API
  7. Return here and enter credentials in the form above
⚠️ Important (Temporary Local Testing):

For local testing only, you can disable Row Level Security (RLS) in Supabase Settings → Tables. For production, set up RLS policies to restrict access by role.
